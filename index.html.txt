// ═══════════════════════════════════════════════════
// BACKGROUND CANVAS — ANIMATED GRID + PARTICLES
// ═══════════════════════════════════════════════════
const canvas = document.getElementById('bg-canvas');
const ctx = canvas.getContext('2d');

function resizeCanvas() {
  canvas.width = window.innerWidth;
  canvas.height = window.innerHeight;
}
resizeCanvas();
window.addEventListener('resize', resizeCanvas);

const GRID_SIZE = 80;
const particles = [];

for (let i = 0; i < 60; i++) {
  particles.push({
    x: Math.random() * window.innerWidth,
    y: Math.random() * window.innerHeight,
    vx: (Math.random() - 0.5) * 0.3,
    vy: (Math.random() - 0.5) * 0.3,
    size: Math.random() * 1.5 + 0.3,
    opacity: Math.random() * 0.5 + 0.1,
    color: Math.random() > 0.6 ? '#00f2fe' : Math.random() > 0.5 ? '#7f00ff' : '#ff00c8',
  });
}

let time = 0;

function drawCanvas() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  time += 0.005;

  // Grid lines
  ctx.strokeStyle = 'rgba(255,255,255,0.025)';
  ctx.lineWidth = 0.5;
  for (let x = 0; x < canvas.width; x += GRID_SIZE) {
    ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, canvas.height); ctx.stroke();
  }
  for (let y = 0; y < canvas.height; y += GRID_SIZE) {
    ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(canvas.width, y); ctx.stroke();
  }

  // Diagonal accent line
  ctx.save();
  ctx.strokeStyle = 'rgba(0,242,254,0.04)';
  ctx.lineWidth = 1;
  for (let i = -20; i < 40; i++) {
    ctx.beginPath();
    ctx.moveTo(i * 100, 0);
    ctx.lineTo(i * 100 + canvas.height * 0.5, canvas.height);
    ctx.stroke();
  }
  ctx.restore();

  // Radial glow spots
  const gradients = [
    { x: canvas.width * 0.15, y: canvas.height * 0.3, color: '0,242,254', r: 350 },
    { x: canvas.width * 0.85, y: canvas.height * 0.6, color: '127,0,255', r: 400 },
    { x: canvas.width * 0.5, y: canvas.height * 0.8, color: '255,0,200', r: 300 },
  ];
  gradients.forEach(g => {
    const grad = ctx.createRadialGradient(g.x, g.y, 0, g.x, g.y, g.r);
    grad.addColorStop(0, `rgba(${g.color},0.06)`);
    grad.addColorStop(1, 'transparent');
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, canvas.width, canvas.height);
  });

  // Particles
  particles.forEach(p => {
    p.x += p.vx; p.y += p.vy;
    if (p.x < 0) p.x = canvas.width;
    if (p.x > canvas.width) p.x = 0;
    if (p.y < 0) p.y = canvas.height;
    if (p.y > canvas.height) p.y = 0;

    ctx.beginPath();
    ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
    ctx.fillStyle = p.color.replace(')', `,${p.opacity * (0.7 + Math.sin(time + p.x) * 0.3)})`).replace('rgb', 'rgba').replace('#', '');

    // Parse hex color manually
    const hex = p.color.replace('#', '');
    const r = parseInt(hex.substring(0, 2), 16);
    const g2 = parseInt(hex.substring(2, 4), 16);
    const b = parseInt(hex.substring(4, 6), 16);
    ctx.fillStyle = `rgba(${r},${g2},${b},${p.opacity * (0.7 + Math.sin(time + p.x) * 0.3)})`;
    ctx.fill();

    // Glow
    ctx.shadowColor = p.color;
    ctx.shadowBlur = 8;
    ctx.fill();
    ctx.shadowBlur = 0;
  });

  requestAnimationFrame(drawCanvas);
}
drawCanvas();

// ANIMATED COUNTER
// ═══════════════════════════════════════════════════
function animateCounter(el, target, duration = 2000, suffix = '') {
  let start = 0;
  const step = target / (duration / 16);
  const update = () => {
    start = Math.min(start + step, target);
    const display = target >= 1000
      ? Math.floor(start).toLocaleString()
      : Math.floor(start);
    el.textContent = display + suffix;
    if (start < target) requestAnimationFrame(update);
  };
  update();
}

// ═══════════════════════════════════════════════════
// INTERSECTION OBSERVER — TRIGGER ANIMATIONS ON SCROLL
// ═══════════════════════════════════════════════════
const observerOpts = { threshold: 0.2 };

// Counter observer
const counterObserver = new IntersectionObserver((entries) => {
  entries.forEach(entry => {
    if (entry.isIntersecting) {
      const el = entry.target;
      const target = parseInt(el.dataset.target);
      animateCounter(el, target, 1800, target >= 1000 ? '+' : target === 30 ? '' : target === 120 ? '+' : '+');
      counterObserver.unobserve(el);
    }
  });
}, observerOpts);

document.querySelectorAll('[data-target]').forEach(el => counterObserver.observe(el));

// Record bar observer
const recordObserver = new IntersectionObserver((entries) => {
  entries.forEach(entry => {
    if (entry.isIntersecting) {
      const widths = ['78%', '91%', '65%'];
      const bars = document.querySelectorAll('.record-bar');
      bars.forEach((bar, i) => {
        setTimeout(() => { bar.style.width = widths[i]; }, i * 150);
      });
      recordObserver.unobserve(entry.target);
    }
  });
}, { threshold: 0.1 });

const recordSection = document.querySelector('#records');
if (recordSection) recordObserver.observe(recordSection);

// Chart bar observer
const chartObserver = new IntersectionObserver((entries) => {
